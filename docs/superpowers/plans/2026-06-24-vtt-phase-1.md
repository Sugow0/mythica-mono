> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Construire une table virtuelle jouable : canvas PixiJS fullscreen avec map uploadée, tokens drag & drop synchronisés en temps réel, fog of war manuel, initiative tracker, chat ancré et présence — tout synchronisé via WebSocket existant.

**Architecture:** Canvas PixiJS (WebGL) monté dans `/sessions/[id]/table`, 4 layers isolés (BackgroundLayer, GridLayer, TokensLayer, FogLayer). Overlays DOM `position: fixed` pour les panneaux. État Svelte 5 `$state` alimenté par WS events → mises à jour ciblées par layer. Backend : nouveau module `maps/` + handlers WS étendus dans le pattern existant `parseCombatCommand`.

**Tech Stack:** Bun · Elysia · Prisma 7 · TypeBox · PixiJS v8 · SvelteKit · Svelte 5 · Tailwind · shadcn-svelte · bun:test

```typescript
// apps/mythica-gameplay/src/maps/maps.service.ts
import { prisma } from "../lib/prisma";

type CreateMapData = {
	nom: string;
	gridType?: "square" | "hex" | "free";
	gridSize?: number;
	gridOffsetX?: number;
	gridOffsetY?: number;
	gridColor?: string;
	gridOpacity?: number;
	widthPx?: number;
	heightPx?: number;
};

export async function getMaps(campaignId: string) {
	return prisma.map.findMany({
		where: { campaignId },
		orderBy: { createdAt: "asc" },
	});
}

export async function getMap(mapId: string, campaignId: string) {
	return prisma.map.findFirst({ where: { id: mapId, campaignId } });
}

export async function createMap(campaignId: string, data: CreateMapData) {
	return prisma.map.create({ data: { campaignId, ...data } });
}

export async function updateMap(
	mapId: string,
	campaignId: string,
	data: Partial<CreateMapData>,
) {
	const existing = await prisma.map.findFirst({ where: { id: mapId, campaignId } });
	if (!existing) return null;
	return prisma.map.update({ where: { id: mapId }, data });
}

export async function deleteMap(mapId: string, campaignId: string) {
	const existing = await prisma.map.findFirst({ where: { id: mapId, campaignId } });
	if (!existing) return false;
	await prisma.map.delete({ where: { id: mapId } });
	return true;
}

export async function setMapFile(mapId: string, fileId: string | null) {
	return prisma.map.update({ where: { id: mapId }, data: { fileId } });
}
```

```typescript
// apps/mythica-gameplay/src/maps/maps.controller.ts
import { Elysia, t } from "elysia";
import { authContext, type CurrentUser } from "../auth/context";
import { ErrorResponse } from "../lib/http";
import { prisma } from "../lib/prisma";
import { sessionRooms } from "../campaigns/session-rooms";
import {
	getMaps, getMap, createMap, updateMap, deleteMap, setMapFile,
} from "./maps.service";
import { CreateMapBody, MapDto, UpdateMapBody } from "./maps.types";

async function resolveCampaignGm(campaignId: string, user: CurrentUser | null) {
	if (!user) return { ok: false as const, status: 401 as const, error: "Non authentifié" };
	const campaign = await prisma.campaign.findFirst({
		where: { id: campaignId, deletedAt: null },
	});
	if (!campaign) return { ok: false as const, status: 404 as const, error: "Campagne introuvable" };
	const member = await prisma.campaignMember.findFirst({
		where: { campaignId, userId: user.id, leftAt: null },
	});
	const isGm = campaign.ownerId === user.id || member?.role === "gm";
	if (!isGm) return { ok: false as const, status: 403 as const, error: "Réservé au MJ" };
	return { ok: true as const, campaign };
}

export const mapsController = new Elysia({ prefix: "/campaigns/:campaignId/maps" })
	.use(authContext)

	.get(
		"/",
		async ({ set, currentUser, params }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			return { success: true as const, maps: await getMaps(params.campaignId) };
		},
		{
			params: t.Object({ campaignId: t.String() }),
			response: {
				200: t.Object({ success: t.Literal(true), maps: t.Array(MapDto) }),
				401: ErrorResponse, 403: ErrorResponse, 404: ErrorResponse,
			},
			detail: { tags: ["Maps"], summary: "Lister les maps d'une campagne" },
		},
	)

	.post(
		"/",
		async ({ set, currentUser, params, body }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			const map = await createMap(params.campaignId, body);
			set.status = 201;
			return { success: true as const, map };
		},
		{
			params: t.Object({ campaignId: t.String() }),
			body: CreateMapBody,
			response: {
				201: t.Object({ success: t.Literal(true), map: MapDto }),
				401: ErrorResponse, 403: ErrorResponse, 404: ErrorResponse,
			},
			detail: { tags: ["Maps"], summary: "Créer une map" },
		},
	)

	.get(
		"/:mapId",
		async ({ set, currentUser, params }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			const map = await getMap(params.mapId, params.campaignId);
			if (!map) { set.status = 404; return { success: false as const, error: "Map introuvable" }; }
			return { success: true as const, map };
		},
		{
			params: t.Object({ campaignId: t.String(), mapId: t.String() }),
			response: {
				200: t.Object({ success: t.Literal(true), map: MapDto }),
				401: ErrorResponse, 403: ErrorResponse, 404: ErrorResponse,
			},
			detail: { tags: ["Maps"], summary: "Détail d'une map" },
		},
	)

	.patch(
		"/:mapId",
		async ({ set, currentUser, params, body }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			const map = await updateMap(params.mapId, params.campaignId, body);
			if (!map) { set.status = 404; return { success: false as const, error: "Map introuvable" }; }
			return { success: true as const, map };
		},
		{
			params: t.Object({ campaignId: t.String(), mapId: t.String() }),
			body: UpdateMapBody,
			response: {
				200: t.Object({ success: true as const, map: MapDto }),
				401: ErrorResponse, 403: ErrorResponse, 404: ErrorResponse,
			},
			detail: { tags: ["Maps"], summary: "Modifier une map" },
		},
	)

	.delete(
		"/:mapId",
		async ({ set, currentUser, params }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			const ok = await deleteMap(params.mapId, params.campaignId);
			if (!ok) { set.status = 404; return { success: false as const, error: "Map introuvable" }; }
			return { success: true as const };
		},
		{
			params: t.Object({ campaignId: t.String(), mapId: t.String() }),
			response: {
				200: t.Object({ success: t.Literal(true) }),
				401: ErrorResponse, 403: ErrorResponse, 404: ErrorResponse,
			},
			detail: { tags: ["Maps"], summary: "Supprimer une map" },
		},
	)

	.post(
		"/:mapId/file",
		async ({ set, currentUser, params, body }) => {
			const access = await resolveCampaignGm(params.campaignId, currentUser);
			if (!access.ok) { set.status = access.status; return { success: false as const, error: access.error }; }
			const existing = await getMap(params.mapId, params.campaignId);
			if (!existing) { set.status = 404; return { success: false as const, error: "Map introuvable" }; }

			return { success: true as const, map, file: fileRecord };
		},
		{
			params: t.Object({ campaignId: t.String(), mapId: t.String() }),
			body: t.Object({ file: t.File({ type: ["image/png", "image/jpeg", "image/webp"], maxSize: "20m" }) }),
			response: {
				200: t.Object({ success: t.Literal(true), map: MapDto, file: t.Any() }),
```

```typescript
// apps/mythica-gameplay/src/campaigns/session-tokens.ts
import { prisma } from "../lib/prisma";
import { resolveSessionAccess } from "./session-access";

export type TokenCommand =
	| { type: "TOKEN_MOVE"; tokenId: string; x: number; y: number }
	| { type: "TOKEN_REMOVE"; tokenId: string }
	| { type: "TOKEN_UPDATE"; tokenId: string; isVisible?: boolean; width?: number; height?: number; label?: string };

export type TokenEvent =
	| { type: "TOKEN_MOVED"; tokenId: string; x: number; y: number }
	| { type: "TOKEN_REMOVED"; tokenId: string }
	| { type: "TOKEN_UPDATED"; tokenId: string; [key: string]: unknown };

export function parseTokenCommand(raw: unknown): TokenCommand | null {
	let data: unknown = raw;
	if (typeof raw === "string") {
		try { data = JSON.parse(raw); } catch { return null; }
	}
	if (!data || typeof data !== "object") return null;
	const cmd = data as Record<string, unknown>;
	switch (cmd.type) {
		case "TOKEN_MOVE":
			if (typeof cmd.tokenId === "string" && typeof cmd.x === "number" && typeof cmd.y === "number") {
				return { type: "TOKEN_MOVE", tokenId: cmd.tokenId, x: cmd.x, y: cmd.y };
			}
			return null;
		case "TOKEN_REMOVE":
			return typeof cmd.tokenId === "string"
				? { type: "TOKEN_REMOVE", tokenId: cmd.tokenId }
				: null;
		case "TOKEN_UPDATE": {
			if (typeof cmd.tokenId !== "string") return null;
			const update: TokenCommand = { type: "TOKEN_UPDATE", tokenId: cmd.tokenId };
			if (typeof cmd.isVisible === "boolean") (update as Record<string, unknown>).isVisible = cmd.isVisible;
			if (typeof cmd.width === "number") (update as Record<string, unknown>).width = cmd.width;
			if (typeof cmd.height === "number") (update as Record<string, unknown>).height = cmd.height;
			if (typeof cmd.label === "string") (update as Record<string, unknown>).label = cmd.label;
			return update;
		}
		default:
			return null;
	}
}

export async function handleTokenCommand(
	sessionId: string,
	userId: string,
	command: TokenCommand,
): Promise<TokenEvent | null> {
	const access = await resolveSessionAccess(sessionId, { id: userId, role: null });
	if (!access.ok) return null;

	switch (command.type) {
		case "TOKEN_MOVE": {
			const token = await prisma.sessionToken.findUnique({ where: { id: command.tokenId } });
			if (!token || token.sessionId !== sessionId) return null;
			await prisma.sessionToken.update({
				where: { id: command.tokenId },
				data: { posX: command.x, posY: command.y },
			});
			return { type: "TOKEN_MOVED", tokenId: command.tokenId, x: command.x, y: command.y };
		}
		case "TOKEN_REMOVE": {
			const token = await prisma.sessionToken.findUnique({ where: { id: command.tokenId } });
			if (!token || token.sessionId !== sessionId) return null;
			await prisma.sessionToken.delete({ where: { id: command.tokenId } });
			return { type: "TOKEN_REMOVED", tokenId: command.tokenId };
		}
		case "TOKEN_UPDATE": {
			const token = await prisma.sessionToken.findUnique({ where: { id: command.tokenId } });
			if (!token || token.sessionId !== sessionId) return null;
			const { tokenId, type: _, ...updates } = command as Record<string, unknown>;
			const updated = await prisma.sessionToken.update({
				where: { id: command.tokenId },
				data: updates as Record<string, unknown>,
			});
			return { type: "TOKEN_UPDATED", tokenId: command.tokenId, ...updates, updatedAt: updated.updatedAt };
		}
	}
}

export async function getSessionTokens(sessionId: string) {
	return prisma.sessionToken.findMany({
		where: { sessionId },
		orderBy: { createdAt: "asc" },
		include: { file: true, characterSheet: { select: { name: true, avatarFileId: true } } },
	});
}
```

```typescript
// apps/mythica-gameplay/src/campaigns/session-map.ts
import { prisma } from "../lib/prisma";
import { isGameMaster, resolveSessionAccess } from "./session-access";

type Point = { x: number; y: number };
type FogState = { mode: "manual" | "dynamic"; revealed: Point[][] };

export type MapCommand =
	| { type: "FOG_PAINT"; polygons: Point[][] }
	| { type: "FOG_CLEAR" }
	| { type: "DICE_ROLL"; formula: string; results: number[]; total: number; isPrivate: boolean };

export type MapEvent =
	| { type: "FOG_STATE"; fog: FogState }
	| { type: "DICE_ROLL"; formula: string; results: number[]; total: number; isPrivate: boolean };

export function parseMapCommand(raw: unknown): MapCommand | null {
	let data: unknown = raw;
	if (typeof raw === "string") {
		try { data = JSON.parse(raw); } catch { return null; }
	}
	if (!data || typeof data !== "object") return null;
	const cmd = data as Record<string, unknown>;
	switch (cmd.type) {
		case "FOG_PAINT":
			if (!Array.isArray(cmd.polygons)) return null;
			return { type: "FOG_PAINT", polygons: cmd.polygons as Point[][] };
		case "FOG_CLEAR":
			return { type: "FOG_CLEAR" };
		case "DICE_ROLL":
			if (
				typeof cmd.formula === "string" &&
				Array.isArray(cmd.results) &&
				typeof cmd.total === "number" &&
				typeof cmd.isPrivate === "boolean"
			) {
				return {
					type: "DICE_ROLL",
					formula: cmd.formula,
					results: cmd.results as number[],
					total: cmd.total,
					isPrivate: cmd.isPrivate,
				};
			}
			return null;
		default:
			return null;
	}
}

function parseFogState(raw: unknown): FogState {
	if (raw && typeof raw === "object") {
		const obj = raw as Record<string, unknown>;
		if (obj.fog && typeof obj.fog === "object") {
			return obj.fog as FogState;
		}
	}
	return { mode: "manual", revealed: [] };
}

export async function handleMapCommand(
	sessionId: string,
	userId: string,
	command: MapCommand,
): Promise<MapEvent | null> {
	// DICE_ROLL : tout membre peut lancer, pas de persistance
	if (command.type === "DICE_ROLL") {
		const access = await resolveSessionAccess(sessionId, { id: userId, role: null });
		if (!access.ok) return null;
		return { type: "DICE_ROLL", ...command };
	}

	// FOG_* : réservé au MJ
	const access = await resolveSessionAccess(sessionId, { id: userId, role: null });
	if (!access.ok) return null;
	if (!isGameMaster(access)({ id: userId, role: null })) return null;

	const session = await prisma.session.findUnique({ where: { id: sessionId } });
	if (!session) return null;

	const currentFog = parseFogState(session.mapState);
	let nextFog: FogState;

	if (command.type === "FOG_CLEAR") {
		nextFog = { mode: currentFog.mode, revealed: [] };
	} else {
		nextFog = { ...currentFog, revealed: command.polygons };
	}

	const mapState = typeof session.mapState === "object" && session.mapState
		? { ...(session.mapState as object), fog: nextFog }
		: { fog: nextFog };

	await prisma.session.update({
		where: { id: sessionId },
		data: { mapState: JSON.stringify(mapState) },
	});

	return { type: "FOG_STATE", fog: nextFog };
}
```

```typescript
import { Graphics } from "pixi.js";
import type { MapConfig } from "../types";

/** Grille hexagonale pointy-top (angle plat en haut et bas). */
export function drawHexGrid(graphics: Graphics, config: MapConfig) {
	const { gridSize, gridColor, gridOpacity, widthPx, heightPx } = config;
	const color = parseInt(gridColor.replace("#", "").slice(0, 6), 16);
	const alpha = gridOpacity;
	const r = gridSize / 2;
	const h = r * Math.sqrt(3);
	const w = r * 2;

	graphics.clear();

	let row = 0;
	for (let y = 0; y < heightPx + h; y += h) {
		const xOffset = (row % 2) * (w * 0.75);
		for (let x = -w; x < widthPx + w; x += w * 1.5) {
			drawHex(graphics, x + xOffset, y, r, color, alpha);
		}
		row++;
	}
}

function drawHex(g: Graphics, cx: number, cy: number, r: number, color: number, alpha: number) {
	const points: number[] = [];
	for (let i = 0; i < 6; i++) {
		const angle = (Math.PI / 180) * (60 * i - 30);
		points.push(cx + r * Math.cos(angle), cy + r * Math.sin(angle));
	}
	g.poly(points).stroke({ color, alpha, width: 1 });
}

export function snapToHex(x: number, y: number, config: MapConfig): { x: number; y: number } {
	// Snap simple au centre de l'hexagone le plus proche (approximation)
	const r = config.gridSize / 2;
	const h = r * Math.sqrt(3);
	const row = Math.round(y / h);
	const xOffset = (row % 2) * r * 0.75;
	const col = Math.round((x - xOffset) / (r * 1.5));
	return { x: col * r * 1.5 + xOffset, y: row * h };
}
```

```typescript
		switch (config.gridType) {
			case "square": drawSquareGrid(this.graphics, config); break;
			case "hex": drawHexGrid(this.graphics, config); break;
			case "free": drawFreeGrid(); break;
		}
	}

	snap(x: number, y: number): { x: number; y: number } {
		if (!this.config) return { x, y };
		switch (this.config.gridType) {
			case "square": return snapToSquare(x, y, this.config);
			case "hex": return snapToHex(x, y, this.config);
			case "free": return snapToFree(x, y, this.config);
		}
	}

	destroy() { this.container.destroy({ children: true }); }
```

```typescript
import type { SessionTokenDto } from "../types";

export type TokenMoveCallback = (tokenId: string, x: number, y: number) => void;

export class TokenSprite {

	private label: Text | null = null;
	public tokenId: string;

	constructor(token: SessionTokenDto, onMove: TokenMoveCallback) {
		this.tokenId = token.id;
		this.container = new Container();
		this.container.x = parseFloat(token.posX) * 100;
		this.container.y = parseFloat(token.posY) * 100;
		this.container.visible = token.isVisible;

		// Placeholder graphique en attendant l'image réelle
		const g = new Graphics();
		g.roundRect(0, 0, token.width * 100, token.height * 100, 8)
			.fill({ color: 0xdc2626, alpha: 0.85 })
			.stroke({ color: 0xffffff, alpha: 0.5, width: 2 });
		this.sprite = g;
		this.container.addChild(g);

		if (token.label) {
			this.label = new Text({
				text: token.label,
				style: { fontSize: 12, fill: 0xffffff, fontFamily: "Public Sans" },
			});
			this.label.anchor.set(0.5, 0);
			this.label.x = (token.width * 100) / 2;
			this.label.y = token.height * 100 + 4;
			this.container.addChild(this.label);
		}

		// Charger l'image si disponible
		if (token.fileId) {
			this.loadImage(token);
		}

		this.makeDraggable(onMove, token);
	}

	private async loadImage(token: SessionTokenDto) {
		try {
			const url = `/api/gameplay/files/${token.fileId}/content`;
			const texture: Texture = await Assets.load(url);
			const s = new Sprite(texture);
			s.width = token.width * 100;
			s.height = token.height * 100;
			s.roundPixels = true;
			this.container.removeChild(this.sprite);
			(this.sprite as Graphics).destroy?.();
			this.sprite = s;
			this.container.addChildAt(s, 0);
		} catch { /* garder le placeholder */ }
	}

	private makeDraggable(onMove: TokenMoveCallback, token: SessionTokenDto) {
		this.container.eventMode = "static";
		this.container.cursor = "pointer";

		let isDragging = false;
		let startMouseX = 0;
		let startMouseY = 0;
		let startContainerX = this.container.x;
		let startContainerY = this.container.y;

		this.container.on("pointerdown", (e) => {
			isDragging = true;
			startMouseX = e.globalX;
			startMouseY = e.globalY;
			startContainerX = this.container.x;
			startContainerY = this.container.y;
			this.container.zIndex = 1000;
		});

		this.container.on("pointermove", (e) => {
			if (!isDragging) return;
			const dx = e.globalX - startMouseX;
			const dy = e.globalY - startMouseY;
			this.container.x = startContainerX + dx;
			this.container.y = startContainerY + dy;
		});

		this.container.on("pointerup", () => {
			if (!isDragging) return;
			isDragging = false;
			this.container.zIndex = 0;
			// Convertir position pixel → coordonnées grille (diviser par 100)
			const gridX = this.container.x / 100;
			const gridY = this.container.y / 100;
			onMove(token.id, gridX, gridY);
		});

		this.container.on("pointerupoutside", () => {
			isDragging = false;
			this.container.zIndex = 0;
		});
	}

	update(token: SessionTokenDto) {
		this.container.x = parseFloat(token.posX) * 100;
		this.container.y = parseFloat(token.posY) * 100;
		this.container.visible = token.isVisible;
	}

	destroy() { this.container.destroy({ children: true }); }
}
```

```typescript
// apps/mythica-front/src/lib/vtt/layers/TokensLayer.ts
import { Container } from "pixi.js";
import type { Application } from "pixi.js";
import type { SessionTokenDto } from "../types";
import { TokenSprite, type TokenMoveCallback } from "../tokens/TokenSprite";

export class TokensLayer {
	private container: Container;
	private sprites: Map<string, TokenSprite> = new Map();
	private onMove: TokenMoveCallback;

	constructor(stage: Container, _app: Application, onMove?: TokenMoveCallback) {
		this.container = new Container();
		this.container.sortableChildren = true;
		this.onMove = onMove ?? (() => {});
		stage.addChild(this.container);
	}

	setOnMove(cb: TokenMoveCallback) { this.onMove = cb; }

	setTokens(tokens: SessionTokenDto[]) {
		// Détruire les sprites qui n'existent plus
		for (const [id, sprite] of this.sprites) {
			if (!tokens.find((t) => t.id === id)) {
				sprite.destroy();
				this.sprites.delete(id);
			}
		}
		// Créer ou mettre à jour
		for (const token of tokens) {
			const existing = this.sprites.get(token.id);
			if (existing) {
				existing.update(token);
			} else {
				const sprite = new TokenSprite(token, this.onMove);
				this.container.addChild(sprite.container);
				this.sprites.set(token.id, sprite);
			}
		}
	}

	syncTokens(tokens: SessionTokenDto[]) { this.setTokens(tokens); }

	addToken(token: SessionTokenDto) {
		const sprite = new TokenSprite(token, this.onMove);
		this.container.addChild(sprite.container);
		this.sprites.set(token.id, sprite);
	}

	removeToken(id: string) {
		const sprite = this.sprites.get(id);
		if (sprite) { sprite.destroy(); this.sprites.delete(id); }
	}

	updateToken(token: Partial<SessionTokenDto> & { id: string }) {
		this.sprites.get(token.id)?.update(token as SessionTokenDto);
	}

	destroy() { this.container.destroy({ children: true }); }
}
```

```typescript
// apps/mythica-front/src/lib/vtt/ws/VttWebSocket.ts
import type { vttState } from "../vtt.svelte";
import { handleTokenEvent } from "./handlers/token.handler";
import { handleFogEvent } from "./handlers/fog.handler";
import { handleMapEvent } from "./handlers/map.handler";
import { handleCombatEvent } from "./handlers/combat.handler";
import { handlePresenceEvent } from "./handlers/presence.handler";
import type { MapConfig, SessionTokenDto, FogState, CombatState } from "../types";

type VttState = typeof vttState;

export class VttWebSocket {
	private ws: WebSocket | null = null;
	private sessionId: string;
	private state: VttState;
	private reconnectDelay = 1000;
	private shouldReconnect = true;
	private reconnectTimer: ReturnType<typeof setTimeout> | null = null;

	constructor(sessionId: string, state: VttState) {
		this.sessionId = sessionId;
		this.state = state;
	}

	connect() {
		// campaignService.sessionSocketUrl() construit l'URL depuis VITE_PUBLIC_SOCKET_API_URL.
		import("$lib/index.js").then(({ campaignService }) => {
			const wsUrl = campaignService.sessionSocketUrl(this.sessionId);
			this._connectTo(wsUrl);
		});
	}

	private _connectTo(wsUrl: string) {
		this.ws = new WebSocket(wsUrl);

		this.ws.onopen = () => {
			this.reconnectDelay = 1000;
		};

		this.ws.onmessage = (e) => {
			try {
				const event = JSON.parse(e.data) as Record<string, unknown>;
				this.dispatch(event);
			} catch { /* trame invalide ignorée */ }
		};

		this.ws.onclose = () => {
			if (!this.shouldReconnect) return;
			this.reconnectTimer = setTimeout(() => {
				this.reconnectDelay = Math.min(this.reconnectDelay * 2, 30_000);
				this.connect();
			}, this.reconnectDelay);
		};
	}

	private dispatch(event: Record<string, unknown>) {
		// Snapshot initial : initialiser tout l'état
		if (event.type === "snapshot") {
			this.state.setTokens((event.tokens as SessionTokenDto[]) ?? []);
			this.state.setActiveMap((event.activeMap as MapConfig) ?? null);
			if (event.mapState && typeof event.mapState === "object") {
				const ms = event.mapState as { fog?: FogState };
				if (ms.fog) this.state.setFog(ms.fog);
			}
			if (event.combatState) this.state.setCombatState(event.combatState as CombatState);
			if (event.participants) this.state.setParticipants(event.participants as string[]);
			return;
		}

		const type = event.type as string;
		if (type.startsWith("TOKEN_")) { handleTokenEvent(this.state, event); return; }
		if (type.startsWith("FOG_")) { handleFogEvent(this.state, event); return; }
		if (type === "MAP_CHANGE") { handleMapEvent(this.state, event); return; }
		if (type === "COMBAT_STATE") { handleCombatEvent(this.state, event); return; }
		if (type.startsWith("PRESENCE_")) { handlePresenceEvent(this.state, event); return; }
	}

	send(event: Record<string, unknown>) {
		if (this.ws?.readyState === WebSocket.OPEN) {
			this.ws.send(JSON.stringify(event));
		}
	}

	disconnect() {
		this.shouldReconnect = false;
		if (this.reconnectTimer) clearTimeout(this.reconnectTimer);
		this.ws?.close();
	}
}
```

```svelte
	import { onDestroy, onMount } from "svelte";
	import { campaignService } from "$lib/index.js";

	let { sessionId, campaignId }: { sessionId: string; campaignId: string } = $props();

	type Message = { id: string; userId: string; content: string; createdAt: string; user: { pseudo: string | null } };
	let messages = $state<Message[]>([]);
	let input = $state("");
	let listEl: HTMLElement;
	let isOpen = $state(true);
	let eventSource: EventSource | null = null;
	// campaignId est nécessaire pour le chat (le chat est lié à la campagne, pas à la session).
	// Le passer comme prop depuis la page, qui dispose de data.campaign.id.

	function scrollBottom() {
		setTimeout(() => { listEl?.scrollTo({ top: listEl.scrollHeight, behavior: "smooth" }); }, 50);
	}

	async function send() {
		if (!input.trim()) return;
		try {
			await campaignService.postMessage(campaignId, input.trim());
			input = "";
		} catch { /* ignorer */ }
	}

	function onKey(e: KeyboardEvent) {
		if (e.key === "Enter" && !e.shiftKey) { e.preventDefault(); send(); }
	}

	onMount(() => {
		eventSource = new EventSource(campaignService.messagesStreamUrl(campaignId), { withCredentials: true });
		eventSource.onmessage = (e) => {
			try {
				const msg = JSON.parse(e.data);
				if (msg.content) { messages = [...messages.slice(-99), msg]; scrollBottom(); }
			} catch { /* ignorer */ }
		};
	});

	onDestroy(() => eventSource?.close());
</script>

<div class="fixed bottom-4 right-4 z-50 flex w-72 flex-col rounded-xl border border-slate-700 bg-slate-900/95 shadow-2xl backdrop-blur">
	<button
		onclick={() => { isOpen = !isOpen; }}
		class="flex items-center justify-between border-b border-slate-700 px-3 py-2 text-xs font-semibold uppercase tracking-wider text-slate-400 hover:text-white"
	>
		<span>Chat</span>
		<span class="text-slate-600">{isOpen ? "▾" : "▸"}</span>
	</button>
	{#if isOpen}
		<ul bind:this={listEl} class="flex max-h-48 flex-col gap-1 overflow-y-auto p-2">
			{#each messages as msg (msg.id)}
				<li class="text-xs">
					<span class="font-medium text-red-400">{msg.user?.pseudo ?? "?"}</span>
					<span class="ml-1 text-slate-300">{msg.content}</span>
				</li>
			{/each}
		</ul>
		<div class="flex gap-1 border-t border-slate-700 p-2">
			<input
				bind:value={input}
				onkeydown={onKey}
				placeholder="Message…"
				class="flex-1 rounded bg-slate-800 px-2 py-1 text-xs text-white placeholder-slate-500 focus:outline-none focus:ring-1 focus:ring-red-600"
			/>
			<button
				onclick={send}
				class="rounded bg-red-600 p-1.5 text-white hover:bg-red-700"
			>
				<SendIcon class="h-3 w-3" />
			</button>
		</div>
	{/if}
</div>
```
